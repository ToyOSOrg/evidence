## Its release workflow installs the toolchain this way
$ gh api repos/Japabu/gbae/contents/.github/actions/toyos-toolchain/action.yml --jq .content | base64 -d
name: ToyOS toolchain
description: Installs the ToyOS Rust toolchain release that matches the toyos-abi crate in Cargo.lock as `toyos`, points cargo at its linker and exports the release tag as TOYOS_TOOLCHAIN

runs:
  using: composite
  steps:
    - shell: bash
      env:
        GH_TOKEN: ${{ github.token }}
      run: |
        sdk=$(grep -A1 '^name = "toyos-abi"$' Cargo.lock | sed -n 's/^version = "\(.*\)"/\1/p')
        tag=$(gh release download --repo ToyOSOrg/ToyOS "toolchain-linux-x86_64-sdk-$sdk" --pattern TOOLCHAIN --output - | sed -n 's/^toolchain //p')
        echo "toolchain: $tag"
        mkdir -p "$RUNNER_TEMP/toyos"
        gh release download --repo ToyOSOrg/ToyOS "$tag" --pattern toyos-toolchain.tar.zst --output - | zstd -dc | tar -C "$RUNNER_TEMP/toyos" -x
        stage2="$RUNNER_TEMP/toyos/x86_64-unknown-linux-gnu/stage2"
        rustup toolchain link toyos "$stage2"
        echo "CARGO_TARGET_X86_64_UNKNOWN_TOYOS_LINKER=$stage2/bin/toyos-ld" >> "$GITHUB_ENV"
        echo "TOYOS_TOOLCHAIN=$tag" >> "$GITHUB_ENV"
        rustc +toyos -vV
EXIT=0
